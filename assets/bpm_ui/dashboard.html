``` ```sql process_nodes SELECT process_node_id, parent_node_id, node_code, element_type, name, description, order_index, execution_mode, subprocess_node_id FROM BPM.process_node WHERE active = true AND coalesce(excluded, false) = false ORDER BY parent_node_id, order_index, node_code; ``` ```sql process_edges SELECT f.process_node_id, src.node_code AS from_node_code, src.name AS from_node_name, dst.node_code AS to_node_code, dst.name AS to_node_name, e.edge_order, e.is_decision, e.condition, e.branch_value FROM BPM.process_edge e JOIN BPM.process_flow f ON f.process_flow_id = e.process_flow_id JOIN BPM.process_node src ON src.process_node_id = e.from_node_id JOIN BPM.process_node dst ON dst.process_node_id = e.to_node_id WHERE f.active = true AND coalesce(f.excluded, false) = false AND coalesce(e.excluded, false) = false ORDER BY f.process_node_id, e.edge_order; ``` ```sql instances SELECT i.process_instance_id, i.process_node_id, i.instance_code, i.status, strftime(i.started_at, '%Y-%m-%d %H:%M') AS started_at, p.name AS process_name, p.version AS process_version FROM BPM.process_instance i JOIN BPM.process_node p ON p.process_node_id = i.process_node_id WHERE coalesce(i.excluded, false) = false ORDER BY i.started_at DESC LIMIT 50; ``` ```sql instance_steps SELECT s.process_instance_id, n.node_code, n.element_type, n.name AS activity_name, s.status, strftime(s.started_at, '%Y-%m-%d %H:%M') AS started_at, strftime(s.completed_at, '%Y-%m-%d %H:%M') AS completed_at FROM BPM.process_instance_step s JOIN BPM.process_node n ON n.process_node_id = s.process_node_id WHERE coalesce(s.excluded, false) = false ORDER BY s.process_instance_id, s.process_instance_step_id;
